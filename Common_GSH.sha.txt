'Show the appropriate tooltip 
sub Page_OnMouseOver	
	dim oObj, iNrPoints,sPointNames(10)
	on error resume next
	set oobj=window.event.srcelement
	if isobject (oObj.onmouseover) then exit sub
	if len(oObj.id)<14 then exit sub
	if instr(ucase(right(oObj.id,16)),"SCTXTSELECTBOX")<>0 then 
		set oShape = oObj.parentnode.parentnode
 		oObj.style.cursor = "hand"
 		if oShape.getcustomproperty("point","tagname")<>"" then 		    
 			for n=0 to oShape.customProperties.Count -1
 			   if lcase(oShape.customProperties.Item(n).type) = "point" and instr(oShape.customProperties.Item(n).name,"Analog")=0 then 
				  iNrPoints = iNrPoints + 1
 				  sPointNames(iNrPoints) =	oshape.customProperties.Item(n).value
    		   end if
 			next
  			If iNrPoints = 1 Then
  	 			if typename(oShape.objects("ConAlarmState"))<>"Nothing" then
					sPoints=oShape.objects("ConAlarmState").datavalue("tagname.name")
				elseif oShape.objects("ScConAlpPV").dataexists("tagname.name") then
					sPoints=oShape.objects("ScConAlpPV").datavalue("tagname.name")
                else 
					sPoints = sPointNames(1)
				end if	
	  			status = "Analog"
				if (not oShape.objects("TxtError") is nothing) and (oShape.objects("TxtEUDesc") is nothing) or (not oShape.objects("TxtPV") is nothing) or (not oShape.objects("RecPV") is nothing) then status = "Status"
      			if (not oShape.objects("TxtError") is nothing) and (not oShape.objects("IndVal") is nothing or not oShape.objects("TxtOP")is nothing) then status = "RegValve"
                if (not oShape.objects("LinScale") is nothing) then status = "Analog"
	 			if status = "Analog" then
					PvTxt = ToolTip_PV(oShape,status)
					SpTxt = ToolTip_SP(oShape,status)
					OpTxt = ToolTip_OP(oShape)
					ModeTxt = ToolTip_Mode(oShape) 
					AlarmTxt = ToolTip_Alm(oShape,"")
					sToolTip= sPoints & vbCR & SPTxt  & PVTxt & OpTxt &  ModeTxt & AlarmTxt
					if AlarmTxt <> "" then
						oObj.title = sToolTip
					else
						oObj.title = mid(sToolTip,1,len(sToolTip)-1)
					end if
	  			elseif status = "RegValve" then
	  				PVTxt = ToolTip_PV(oShape,status)
					OPTxt = ToolTip_OP(oShape)
					AlarmTxt = ToolTip_Alm(oShape,"")
					sToolTip= sPoints & vbCR & PVTxt  & OPTxt & AlarmTxt
					if AlarmTxt <> "" then
						oObj.title = sToolTip
					else
						oObj.title = mid(sToolTip,1,len(sToolTip)-1)
					end if
	  			else
					AlarmTxt = ToolTip_Alm(oShape,"")
					sToolTip = sPoints & vbcr &  PVTxt &  AlarmTxt 
					if AlarmTxt <> "" then
						oObj.title = sToolTip
					else
						oObj.title = mid(sToolTip,1,len(sToolTip)-1)
					end if
	  			end if
  			Elseif iNrPoints > 1 Then
    			sPoints = ucase(sPointNames(1)) & vbcr
				For i=2 to iNrPoints
	  				if sPointNames(i)<>sPointNames(1) Then 
	  					sPoints = sPoints & vbcr & ucase(sPointNames(i)) 
						AlarmTxt = ToolTip_Alm(oShape, i-1)
						if AlarmTxt <> "" and typename(oShape.objects("ConAlarmState" & i-1))<>"Nothing" then
							if oShape.objects("ConAlarmState" & i-1).style.visibility = "inherit" then sAlm = AlarmTxt
						end if
       				end if
     			Next 
				if sAlm<>"" then sPoints = sPoints & vbcr & vbcr & sAlm
				oObj.title= sPoints 
  			End If
			if oShape.getcustomproperty("Text","FaceplateFile")="-" then 
				oObj.style.cursor="default"
			else
				oObj.style.cursor="hand"
			end if
			if oShape.getcustomproperty("text","faceplatefile")="-" then oShape.objects("sctxtselectbox").style.cursor="default"
   		end if
 	elseif ucase(right(oObj.id,14))="SCTXTSELECTION" then 
    	set oShape=oObj.parentnode.parentnode
		oShape.objects("ScTxtSelection").style.cursor="hand"
 	end if
end sub

'Get the PV Text shown in tooltip. Called within Page_OnMouseOver
function ToolTip_PV (oShape,PointType)
	on error resume next
	select case PointType
		case "Analog"
			sPV = oShape.getcustomproperty("parameter","cp_pv")
			sPos = instr(sPV,".")
			if spos>0 then sPV = mid(sPV,sPos+1,len(sPv)-sPos)
    		EUDesc = oShape.objects("ScConAlpPV").Datavalue("tagname.cp_eudesc")
			PVFormat = oShape.getcustomproperty("Text","cp_pvformat")
			if PVFormat="" then PVFormat = 2
			PVFormat = oShape.objects("ScConAlpPV").Datavalue("tagname.cp_pvformat")
			PV = oShape.objects("ScConAlpPV").value
			If isnumeric(PV) and sPV<>"" Then 
				PVTxt = ucase(sPV) & "="  & FormatNumber (PV, PVFormat) & " " & EuDesc & vbcr
			elseif sPV<>"" then
				PVTxt =ucase(sPV) & "="  & PV & vbcr
			end if	
		case "RegValve"
			sPV = oShape.getcustomproperty("parameter","cp_pv")
			sPos = instr(sPV,".")
			if spos>0 then sPV = mid(sPV,sPos+1,len(sPv)-sPos)
	    	PV = oShape.objects("ScConAlpPV").value
	    	If sPV<>"" Then PVTxt = ucase(sPV) & "="  & FormatNumber (PV, 0) & " " & "%" & vbcr			
	end select
	ToolTip_PV= PVTxt
end function

'Get the SP Text shown in tooltip. Called within Page_OnMouseOver
function ToolTip_SP (oShape,PointType)
	on error resume next
	select case PointType
		case "Analog"
			sSP = oShape.getcustomproperty("parameter","cp_sp")
			sPos = instr(sSP,".")
			if spos>0 then sSP = mid(sSP,sPos+1,len(sSP)-sPos)
			if sSP = "" and oShape.getcustomproperty("parameter","cp_dev") <>"" then sSP = "SP"		
			SP=""
			SP = oShape.objects("ConAlpSP").value
			SP = oShape.objects("ScConAlpPV").datavalue("Tagname.cp_sp")
			SP = oShape.objects("ScConAlpPV").value - oShape.objects("ScConAlpPV").Datavalue("tagname.cp_dev")
			PVFormat = oShape.getcustomproperty("Text","cp_pvformat")
			if PVFormat="" then PVFormat = 2
			PVFormat = oShape.objects("ScConAlpPV").Datavalue("tagname.cp_pvformat")
			EUDesc = oShape.objects("ScConAlpPV").Datavalue("tagname.cp_eudesc")		
			If isnumeric(SP) Then SPTxt = vblf & ucase(sSP) & "="  & FormatNumber (SP, PVFormat) & " " & EuDesc & vbcr
	end select
	ToolTip_SP= SPTxt
end function

'Get the OP Text shown in tooltip. Called within Page_OnMouseOver
function ToolTip_OP(oShape)',PointType)
	on error resume next
			sOP = oShape.getcustomproperty("parameter","cp_op")
			sPos = instr(sOP,".")
			if spos>0 then sOP = mid(sOP,sPos+1,len(sOP)-sPos)
        	PVFormat=1
			OP=""
			OP = oShape.objects("ScConAlpPV").value
			OP = oShape.objects("ConAlpOP").value
			OP = oShape.objects("ConIndOP").value
			OP = oShape.objects("ScConAlpOP").datavalue("Tagname.cp_op")
	    	If sOP<>"" Then OPTxt = ucase(sOP) & "="  & FormatNumber (OP, PVFormat) & " " & "%" & vbcr
	ToolTip_OP = OPTxt
end function

'Get the Mode Text shown in tooltip. Called within Page_OnMouseOver
function ToolTip_Mode(oShape)
	on error resume next
	ModeTxt = oShape.objects("ScConAlpPV").Datavalue("tagname.cp_mode")
	if len(ModeTxt)>0 Then ToolTip_Mode = "Mode= " & ModeTxt &vbcr
end function

'Get the Alarm Text shown in tooltip. Called within Page_OnMouseOver
function ToolTip_Alm (oShape,index)
	on error resume next
	if typename(oShape.objects("ConAlarmState" & index))<> "Nothing" then 
		AlarmTxt = oShape.objects("ConAlarmState" & index).title
		IF len(AlarmTxt)<>0 and instr(AlarmTxt,"Quality")=0 Then 
			ToolTip_Alm = "Alarm= " & AlarmTxt
		elseif index="" then 
			ToolTip_Alm = "No Alarm "
		else
			 ToolTip_Alm=""
		end if
	end if
end function

'Focus ConAlarmState if an alarm is present or ScConAlpPV when there is no alarm.
sub ObjSel
	dim oShape
	on error resume next
	set oShape=document.getelementbyID(page.getattribute("lastselection")).parentnode.parentnode
	set oAlarmIcon = nothing
  	for each oAlarmState in oShape.children(0).all
    	if instr(oAlarmState.classname, "alarmstate")>0 and (oAlarmState.Style.Visibility = "inherit" or oAlarmState.Style.Visibility = "")  then
			if instr(oAlarmState.title,"Unack") or instr(ucase(page.getattribute("lastselection")),"ALM_SCTXTSELECTBOX")>0 and oAlarmState.title<>"" then
				set oAlarmIcon = oAlarmState	
			end if						
    	end if
  	next
  	if oAlarmIcon is nothing Then
		for each oAlphaNum in oShape.all.tags("TEXTAREA")
			if (oAlphaNum.alarm and &H400)=&H400 then 
				set oAlarmIcon=oAlphaNum
				exit for
			end if	             
  		next
    end if 
  	if oAlarmIcon is nothing Then
    	oshape.objects("ScConAlpPV").focus
  	else
    	oAlarmIcon.focus
  	end if
end sub

    	    sPoints = sPoints & "&" & oshape.customProperties.Item(n).name & "=" & oshape.customProperties.Item(n).value

			end if

	   end if

    end if 
    iYoke=ucase(page.getstyleclassproperty("Enable_Yoking","key"))
	if iYoke="ON" and not isnull(page.getattribute("GSH_Yoking")) then call DspYoking(oSource) 
end sub

'Change the ScTxtSelectBox color when selected
sub SelectBox_Color(oSource)
	dim soldsel,skey,bYok, status, oShape 
	on error resume next
	soldsel=page.getattribute("lastselection")
	set oShape=oSource.parentnode.parentnode
	if not isnull(soldsel) then page.all.item(soldsel).styleclass = "ScTxtSelectBox_NotSelected"
    page.setattribute "lastselection",osource.id
	if ucase(right(oSource.id,18)) <> "ALM_SCTXTSELECTBOX" then osource.styleclass = "ScTxtSelectBox_Selected"
	skey=page.getstyleclassproperty("Enable_Assoc_Focus","key")
	if ucase(skey)="ON" and not isnull(page.getattribute("GSH_Yoking")) then
	    call assoc_focus(oSource,soldsel)
    if not isnull(soldsel) then
		    set oShape_Old=document.getelementbyid(soldsel).parentnode.parentnode
			status="Analog"
			if (not oShape_old.objects("TxtError") is nothing) and (oShape_old.objects("TxtEUDesc") is nothing) or (not oShape_old.objects("TxtPV") is nothing) or (not oShape_old.objects("RecPV") is nothing) then status = "Status"
	    	if (not oShape_old.objects("TxtError") is nothing) and (not oShape_old.objects("IndVal") is nothing or not oShape_old.objects("TxtOP")is nothing) then status = "RegValve"
	    	if (not oShape_old.objects("LinScale") is nothing) then status = "Analog"  		  
			select case status
				case "Analog" if oShape_old.Objects("RecFrame").styleclass="RecFrameInPrincipalFocus" then oShape_old.Objects("RecFrame").styleclass="RecFrame"
				case "Status", "RegValve" if oShape_old.Objects("RecFrame").styleclass="RecFrameInPrincipalFocus" then oShape_old.Objects("RecFrame").styleclass="RecFrameNoInFocus"
	        end select	
		end if 
		oShape.Objects("RecFrame").styleclass="RecFrameInPrincipalFocus"
	end if
	if err.number<>0 and window.external.environment = 0 then window.external.LogMessage " SelectBox_Color ->" & oSource.id & " " & err.description
end sub

'Determine the most importatnt alarm. Used in shapes with multiple tagnames
Sub Multiple_Alarm_Icon(oSource)
	Dim sMostImportantAlarm, sAlarmState, iThisImportancy, iMostImportant, oAlarmState, iAlarmCount, iInActiveAlarmCount, oAlm
	On error resume next
	iMostImportant=0
	for each oAlarmState in oSource.parentnode.children
 		if instr(oAlarmState.classname, "alarmstate")>0 then
 			iAlarmCount = iAlarmCount + 1
    		iThisImportancy = 0
    		iThisImportancy = 24+oAlarmState.Acknowledged*16 + oAlarmState.Priority*2 + oAlarmState.RTN*8
    		if iThisImportancy  = 16 then iThisImportancy = 8
    		if iThisImportancy >iMostImportant then 
       			sMostImportantAlarm = oAlarmState.id
       			iMostImportant = iThisImportancy
    		end if
			if iThisImportancy = 0 then iInActiveAlarmCount = iInActiveAlarmCount + 1 
 		end if
	next
	for each oAlarmState in oSource.parentnode.children
   		if instr(oAlarmState.classname, "alarmstate")>0 then 
      		if oAlarmState.id = sMostImportantAlarm or iAlarmCount = iInActiveAlarmCount then
 	  	 		oAlarmState.Style.Visibility = "inherit"
				set oAlm = oAlarmState
      		else
         		oAlarmState.Style.Visibility = "hidden"
      		end if
    	end if
	next
 call All_ObjAlm_OnUpdate(oAlm)
 if err.number<>0 and window.external.environment = 0 then window.external.LogMessage " Multiple_Alarm_Icon ->" & oSource.id & " " & err.description
End Sub

			oSymbLow.linecolorblink=false

			oSymbLow.fillcolorblink=false

	end if
	if iTotalHighAlarms>0 then
		oSymbHigh.style.visibility = "inherit"
	   	oSymbHigh.fillcolor = AlarmStateIconFactory.HighColor
		oSymbHigh.linecolor = AlarmStateIconFactory.HighColor

			oSymbHigh.fillcolorblink=false
  		elseif iTotalHighUnackAlarms>0 then
			oSymbHigh.fillcolorblink=true
