'--------------------------------------
'RegCntl - Mode; Redtag; Hialm Type; EUDesc; 
'--------------------------------------
sub HPM_RegCntl_Mode_Hialm_RedTag_EUDesc_OnUpdate(oSource,iLen)
   dim sMode, oMode, sNMode, iAlmEnbState, iRedTag, oShape,oPIDAlm, oInitWindup, iInitMan,iArwop, iNModeAttr, iModeAttr
   on error resume next
   set oShape=oSource.parentnode.parentnode 
   if typename(oShape.objects("TxtMode"))<>"Nothing" then
	   set oMode=oShape.objects("TxtMode")
	   if oshape.getcustomproperty("parameter","cp_mode")<>"" then
	      sMode=oSource.DataValue("tagname.cp_mode")
	      oMode.innertext=left(smode,ilen)
	   end if
	   if oshape.getcustomproperty("parameter","cp_nmode")<>"" then
	      sNMode=oSource.DataValue("tagname.cp_nmode")
		  if sMode=sNMode then
	      	oMode.styleclass="TxtMode_in_NMode"
	      else 
	      	oMode.styleclass="TxtMode_not_in_NMode"
	      end if 
	   end if
	   if oshape.getcustomproperty("parameter","cp_nmodeAttr")<>"" then
		  iNModeAttr=oSource.DataValue("tagname.cp_nmodeAttr")
		  iModeAttr=oSource.DataValue("tagname.cp_modeAttr")	
		  if iNModeAttr<>0 then
		   	  if iModeAttr=iNModeAttr then
		      	oMode.linecolor="transparent"
		      else 
		      	oMode.linecolor=oMode.getstyleproperty("mAttrProgF") 
				oMode.linestyle=1
			    oMode.linewidth=oMode.getstyleproperty("lWidth")
		      end if 
			  if iModeAttr=2 then oMode.textcolor = oMode.getstyleproperty("mAttrProgL")
           end if
	   end if		
   end if
   if oshape.getcustomproperty("parameter","cp_highal")<>"" then 
		oShape.objects("TxtHiAlmType").innertext = AlmTp(oSource.DataValue("tagname.cp_highal"))
   end if
   if oshape.getcustomproperty("parameter","cp_redtag")<>"" then 
  	 iRedTag=oSource.DataValue("tagname.cp_redtag")
  	 if iRedTag=1 then
  	    oShape.objects("WdgRedTag").styleclass="WdgRedTag_On"
 	 else
	    oShape.objects("WdgRedTag").styleclass="WdgRedTag_Off"
   	 end if
   end if
   if oshape.getcustomproperty("parameter","cp_initman")<>"" then
      set oInit = oShape.objects("TxtInit") 
	  set oWindup = oShape.objects("LinWR")
	  iInitMan = oSource.datavalue ("tagname.cp_initman")
	  oWindup.styleclass = "LinWR_No_WindupReset"
	  	if iInitMan = 1 then
			oInit.styleclass = "Init_Active"
			oWindup.styleclass = "LinWR_No_WindupReset"
		else
			oInit.styleclass = "Init_Inactive"
			if oshape.getcustomproperty("parameter","cp_sptvstate")<>"" then
				iTVstate=oSource.datavalue("Tagname.cp_sptvstate")
				iTV=oSource.datavalue("Tagname.cp_sptv")
				if iTVState=2 then
					if typename(oShape.objects("ConAlpSP"))<>"Nothing" then
						rSP=oShape.objects("ConAlpSP").value	
                    else
                        if oshape.getcustomproperty("parameter","cp_dev")<>"" then
							rSP=oSource.value-oSource.datavalue("tagname.cp_dev")
                        else
							rSP=oSource.datavalue("tagname.cp_sp")
						end if
					end if 	
				    if iTV>rSP then oWindup.styleclass="LinWR_Ramp_HI"
					if iTV<rSP then oWindup.styleclass="LinWR_Ramp_LO"
                else
					oWindup.styleclass="LinWR_No_WindupReset"
				end if
            end if
			if oshape.getcustomproperty("parameter","cp_arwop")<>"" then
				iArwop = oSource.datavalue ("tagname.cp_arwop")
				select case iArwop
				     case 1 oWindup.styleclass="LinWR_Windup_Hi"
				     case 2 oWindup.styleclass="LinWR_Windup_LO"
				     case 3 oWindup.styleclass="LinWR_Windup_HILO"
				end select
            end if  
		end if
   end if
   if oShape.getcustomproperty("Text","cp_TagnameEU")="" then oShape.objects("TxtEUDesc").innertext = oSource.DataValue("tagname.cp_eudesc")
   if err.number <>0 and window.external.environment = 0 then window.external.LogMessage " HPM_RegCntl_Mode_Hialm_RedTag_EUDesc_OnUpdate ->" & oSource.id & " " & err.description 
end sub

sub HPM_Numeric_AlpPVFormat_OnUpdate(oSource,data_src)
   dim sFormat,oShape, nNumericFormat
   on error resume next
   set oShape=oSource.parentnode.parentnode
   if ucase(data_src)="SD" then
	  if oShape.getcustomproperty("Parameter","cp_pvformat")<>"" then nNumericFormat= oSource.DataValue("tagname.cp_pvformat")
   else
      nNumericFormat= trim(oshape.getcustomproperty("Text","cp_pvformat"))
   end if
   if IsNumeric(nNumericFormat) and instr(nNumericFormat,".")=0 then
	  sFormat="%." & nNumericFormat & "f"
	  if typename(oshape.objects("ScConAlpPV"))<>"Nothing" then oShape.objects("ScConAlpPV").numericdisplayformat=sFormat
	  if typename(oshape.objects("ConAlpSP"))<>"Nothing" then oShape.objects("ConAlpSP").numericdisplayformat=sFormat
   else 
   	  window.external.LogMessage "HPM_Numeric_AlpPVFormat_OnUpdate ->" & oSource.id & " " & "Invalid PV Format: " & nNumericFormat 
   end if
   if err.number <>0 and window.external.environment = 0 then window.external.LogMessage "HPM_Numeric_AlpPVFormat_OnUpdate ->" & oSource.id & " " & err.description 
end sub

sub HPM_RegCntl_AlpDev_OnUpdate(oSource)
   on error resume next
   dim rDev, rRange,oShape
   set oShape=oSource.parentnode.parentnode
   if oShape.getcustomproperty("Parameter","cp_dev")<>"" then rDev=oSource.DataValue("tagname.cp_dev") 
   if oShape.getcustomproperty("Parameter","cp_pveuhi")<>"" and oShape.getcustomproperty("Parameter","cp_pveulo")<>"" then
   		rRange=0.05*(oSource.DataValue("tagname.cp_pveuhi")-oSource.DataValue("tagname.cp_pveulo"))
   end if
   rRange=csng(oShape.getcustomproperty("Real","RangeDev"))
   if oShape.getcustomproperty("Parameter","cp_advdevalm.tp")<>"" then 
   	  if instr(oSource.DataValue("tagname.cp_advdevalm.tp"),"N")=0 then	rRange=csng(oSource.DataValue("tagname.cp_advdevalm.tp"))
   end if 
   oShape.objects("IndDev").value=(rDev*100/rRange)
   if err.number<>0 and window.external.environment = 0 then window.external.LogMessage "HPM_RegCntl_AlpDev_OnUpdate ->" & oSource.id & " " & err.description 
end sub

'--------------------------------------

sub HPM_Bar_OnUpdate(oSource)
    dim oShape,cpRangeHI,cpRangeLO,rangeHI,rangeLO, PriorityLL,PriorityL,PriorityH,PriorityHH
	dim Ind_HH,Ind_H,Ind_L,Ind_LL,sColor,PV,SP,TripHH,TripH,TripL,TripLL
	on error resume next
	set oShape=oSource.parentnode.parentnode
	if oshape.getcustomproperty("parameter","cp_mode")<>"" then
		sMode=oSource.datavalue("tagname.cp_mode")
		oShape.objects("TxtMode").innertext=left(sMode,1)
	end if
	if oshape.getcustomproperty("parameter","cp_nmode")<>"" then
      sNMode=oSource.DataValue("tagname.cp_nmode")
	  if sMode=sNMode then
      	oShape.objects("TxtMode").styleclass="TxtMode_in_NMode"
      else 
      	oShape.objects("TxtMode").styleclass="TxtMode_not_in_NMode"
      end if 
    end if
    rangeHI=oSource.datavalue("Tagname.cp_pveuhi")
    rangeLO=oSource.datavalue("Tagname.cp_pveulo")
	cpRangeHI=oShape.getcustomproperty("Real","cp_rangeHI")
	if cpRangeHI="" then cpRangeHI=rangeHI
    cpRangeLO=oShape.getcustomproperty("Real","cp_rangeLO")
	if cpRangeLO="" then cpRangeLO=rangeLO
    PV=oSource.value
	if csng(PV)<csng(cpRangeLO) then PV=cpRangeLO
	if csng(PV)>csng(cpRangeHI) then PV=cpRangeHI
    oShape.objects("IndPV").value=(PV-cpRangeLO)/(cpRangeHI-cpRangeLO)*100
    if oshape.getcustomproperty("parameter","cp_sp")<>"" and typename(oShape.objects("IndSP"))<>"Nothing" then
		SP=oSource.datavalue("tagname.cp_sp")

    dim oShape,cpRange,DevHI,DevLO,PriorityDevHI,PriorityDevLO
