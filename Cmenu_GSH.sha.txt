'Open help file using Windows Explorer, if configured in cm_file custom property
sub help ()
	on error resume next
	dim oshape, spath, cm_file
	set oshape = document.getelementbyid(page.getattribute("lastselection")).parentnode.parentnode
	cm_file = oshape.getcustomproperty("text","cm_file")
	if instr(cm_file,"\")>0 then
 		window.external.shell "explorer.exe " & cm_file,1
	else
 		spath = page.document.url
 		spath=mid (spath,8,len(spath))
 		spath=mid (spath,1,instrrev(spath,"\"))  & "helpfiles\" & cm_file
 		spath = Replace(spath,"%20"," ")
 		window.external.shell "explorer.exe " & spath,1
	end if
	if err.number <>0 and window.external.environment = 0 then window.external.LogMessage "help ->" & oShape.ID & " " & err.description 
end sub

'View Trend Configuration if enabled in View_Trd_Load
sub view_trd
	dim oShape
	on error resume next
	set oshape=document.getelementbyid(page.getattribute("Trend_Sup_GSH"))
	if oshape.objects("PbHide").style.visibility="visible" then oshape.objects("PbHide").click
	oshape.style.zindex=5000
	iPageWidth = page.style.pixelwidth
	iSelectedShapeX = document.activeElement.parentnode.parentnode.style.pixelleft
	if iSelectedShapeX > iPageWidth/2 then
		iTableLeft = 20
	else
		iTableLeft = iPageWidth - 420
	end if
	oshape.style.pixelleft=iTableLeft
	oshape.style.pixeltop=80
	oshape.objects("trdtbl").style.visibility="visible"
	oshape.objects("trdtbl").style.pixelwidth=400
	oshape.objects("PbHide").style.visibility="visible"
	oshape.objects("PbHide").style.pixelleft=oshape.objects("PbHide").style.pixelleft+336
	if oShape.objects("alpisprotected").value = 1 and window.external.securitylevel<4 then oshape.objects("RecProtected").style.pixelwidth=400	
	if err.number <>0 and window.external.environment = 0 then window.external.LogMessage "view_trd ->" & oShape.ID & " " & oSource & " " & err.description

' Invoke trend popup. called within sp_systrdsel_sm.xml
sub trd_popup
	on error resume next
	set oShape = document.getElementById(page.getattribute("Trend_Sup_GSH"))

' subroutines added in 2009
sub Toggle_SCM_NameDesc (pointtype)
	on error resume next
	dim oshape, custgrp, sDesc
	set oshape = document.getelementbyid(page.getattribute("lastselection")).parentnode.parentnode
	set oText = oShape.objects("TxtNameSeq")
	select case pointtype
		case "cda"  sDesc = "desc"
		case "hpm", "hg"  sDesc = "ptdesc"
		case "scd" sDesc = "description"
	end select
	if oShape.getcustomproperty("Parameter","cp_BlockName")<>"" then
		sName = oShape.getcustomproperty("Parameter","cp_BlockName")
		sDVDesc = "tagname." & sName & "."  & sDesc
	else
		if typename(oShape.objects("ConAlarmState"))<>"Nothing" then
			sName=oShape.objects("ConAlarmState").datavalue("tagname.name")
		elseif oShape.objects("ScConAlpPV").dataexists("tagname.name") then
			sName=oShape.objects("ScConAlpPV").datavalue("tagname.name")
	    else 
			sName = oShape.getcustomproperty("Point","Tagname")
		end if
		sDVDesc = "tagname." & sDesc
	end if
	if oText.innertext = sName then
		oText.innertext = oShape.objects("ScConAlpPV").datavalue(sDVDesc)
	else
		oText.innertext =sName
	end if
	if err.number <>0 and window.external.environment = 0 then window.external.LogMessage "Toggle_SCM_NameDesc ->" & oShape.ID & " " & err.description 
end sub

sub TagnameYoking()
	dim oShape,Database,sDB,sTagname,sql,rs,i
	on error resume next
	i=1
	set oshape = document.getelementbyid(page.getattribute("lastselection")).parentnode.parentnode
	sDB = page.GetStyleClassProperty("DatabaseFile","Filename")

	set Database = CreateObject("ADODB.Connection")
	Database.Open("DRIVER={Microsoft Access Driver (*.mdb)}; DBQ=" _
	& sDB )

	if typename(oShape.objects("ConAlarmState"))<>"Nothing" then
		sTagname=oShape.objects("ConAlarmState").datavalue("tagname.name")
	elseif oShape.objects("ScConAlpPV").dataexists("tagname.name") then
		sTagname=oShape.objects("ScConAlpPV").datavalue("tagname.name")
    else 
		sTagname = oShape.getcustomproperty("Point","Tagname")
	end if	
	 
	sql = "select * from TagnameYoking WHERE Tagname Like '" & sTagname & "'"
  set rs = database.Execute(sql)
  if rs.EOF = true then
  		window.external.Application.messagezonetext "Yoking not configured" 
  else
		while not rs.EOF and i < 4
			if not isnull(rs.fields(1).value) then 
				window.external.Application.currentpage = rs.fields(1).value
				msgbox rs.fields(1).value
			end if
			rs.movenext
			i=i+1
		wend
  end if
database.close
set database = nothing

if err.number <>0 and window.external.environment = 0 then window.external.LogMessage "yoking_load ->" & oShape.ID & " " & oSource & " " & err.description 
end sub
